# Base condivisa da tutti i servizi della suite.
#
# Qui va solo ciò che serve davvero a tutti. Le dipendenze specifiche di un
# motore (diffusers per Dream, i pacchetti di ComfyUI per Musica e Foto, quelli
# di SoulX per IoDigitale) stanno nel requirements del rispettivo servizio: così
# chi installa solo il Visualizer non si porta dietro mezza libreria di computer
# vision.
#
# torch NON è qui: si installa a parte con --torch-backend=auto, perché la wheel
# giusta dipende dal driver CUDA della macchina.
#
# **I numeri esatti non stanno qui, stanno in `versioni.txt`**, che è il file di
# vincoli applicato a *ogni* installazione della suite — comprese quelle di
# ComfyUI e dei nodi custom, che non abbiamo scritto noi. Questo file dice
# **cosa** serve; quello dice **in che versione**, una volta per tutti.

# Ogni servizio parla HTTP con lo shell ed espone /health e /shutdown.
fastapi>=0.121.0
uvicorn[standard]>=0.38.0
websockets>=15.0
python-multipart>=0.0.20

# Scaricamento dei modelli: `snapshot_download` serve per i repo HuggingFace
# interi (SD-Turbo, SoulX).
#
# **Qui c'era un tetto `<1.0`, ed è stato tolto il 19 agosto 2026.** La ragione
# scritta accanto — «transformers pretende huggingface-hub<1.0» — era vera con
# transformers 4, ed è diventata esattamente il contrario con la 5, che pretende
# `>=1.0`. Il tetto quindi non proteggeva più niente: faceva **rimbalzare** le
# due librerie a ogni installazione, e da un rimbalzo finito male sono nate
# quattro app che non si aprivano. Adesso il numero è fermo in `versioni.txt`.
huggingface-hub>=1.28
safetensors>=0.6.0

# Tokenizzatore del traduttore italiano→inglese. Senza, MarianMT funziona lo
# stesso ma con una segmentazione peggiore e un avviso a ogni avvio.
sacremoses>=0.1.1

# Lettura della VRAM per l'arbitro della GPU.
nvidia-ml-py>=13.0

# FFmpeg, per tutta la suite.
#
# **Perche' e' salito qui dalla 0.8.0.** Stava nei requisiti di DaProdDream e di
# DaProdIoDigitale, che lo usano per registrare. Ma FFmpeg serve a tre cose che
# non sono di nessuna scheda: il fotogramma di anteprima di un video, la
# copertina cucita dentro un brano, la transcodifica del Visualizer. Chi non
# aveva installato quelle due schede non aveva FFmpeg, e quelle tre cose non
# funzionavano — **senza che da nessuna parte ci fosse scritto perche'**. Le
# anteprime dei video nere sono state questo per tre versioni di fila.
#
# Non e' un doppione di quello di sistema: `findFfmpeg` guarda prima il PATH e
# usa questo solo se sul computer non ce n'e' uno. E non cambia niente sulla
# licenza: il binario lo scarica pip sul computer di chi installa, come
# ComfyUI. La suite resta MIT e non imbarca niente di GPL.
imageio-ffmpeg>=0.6.0
